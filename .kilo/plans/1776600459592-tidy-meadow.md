### 6. core/history.js (NEW - wrapper)

Wrap the existing `HistoryManager` class:

```javascript
import { HistoryManager } from '../historyManager.js';
import { state } from './state.js';

const historyManager = new HistoryManager();

historyManager.setNodeHandlers(
  () => state.nodes,
  (newNodes) => { state.nodes = newNodes; }
);
historyManager.setBoxHandlers(
  () => ({ 
    x: state.spriteBox.x, y: state.spriteBox.y,
    width: state.spriteBox.width, height: state.spriteBox.height,
    scale: state.view.charScale
  }),
  (box) => { Object.assign(state.spriteBox, box); state.view.charScale = box.scale; }
);

export function saveHistory() {
  historyManager.saveState();
}

export function undo() {
  if (historyManager.undo()) {
    state.nodes = historyManager.history[historyManager.currentIndex]?.nodes || state.nodes;
  }
}

export function redo() {
  if (historyManager.redo()) {
    state.nodes = historyManager.history[historyManager.currentIndex]?.nodes || state.nodes;
  }
}

export function canUndo() { return historyManager.currentIndex > 0; }
export function canRedo() { return historyManager.currentIndex < historyManager.history.length - 1; }
```

// --- BODY LOADING ---
async function loadBody(bodyType) {
  const bodyData = await loadBodyDefinition(bodyType);
  if (!bodyData) return;
  
  state.nodes = getCurrentNodes();
  state.bones = getCurrentBones();
  state.constraints = getCurrentConstraints();
  state.meta.bodyType = bodyType;
  state.meta.height = bodyData.height || 180;
  if (!state.meta.name) state.meta.name = bodyData.name;
  
  // Compute hierarchy from bones
  if (state.bones.length > 0) {
    const rootNode = state.nodes.find(n => n.id === 'pelvis') ? 'pelvis' : state.bones[0][0];
    const tree = hierarchy.buildTree(state.bones, rootNode);
    const hierarchyMap = hierarchy.computeAllDescendants(tree.children);
    state.currentHierarchy = hierarchyMap;
    state.currentPelvisChildren = hierarchyMap[rootNode] || [];
    
    // Foot nodes: terminal effectors
    const isHuman = ['adult-male', 'adult-female', 'child'].includes(bodyType);
    state.currentFootNodes = [];
    Object.keys(tree.children).forEach(node => {
      const children = tree.children[node];
      if (!children || children.length === 0) {
        if (node.startsWith('foot')) state.currentFootNodes.push(node);
        else if (node.startsWith('hand') && !isHuman) state.currentFootNodes.push(node);
        else if (node.includes('paw')) state.currentFootNodes.push(node);
      }
    });
    state.currentGroundY = hierarchy.calculateGroundY(state.nodes);

  anim.initFrames(state);

  timeline.updateFrameBadge();
  sidebar.updateAnimationList(bodyType);
  importExport.renderSavedList();
  render();
}

## Interaction Details

### Mouse Events
- `mousedown`:
  - Left click on node → startDrag(node)
  - Left click on pelvis special handling (drag all pelvis children)
  - Left click on bounding box handle → startDragBox(handle)
  - Left click on empty → startPan OR deselect
  - Middle click → startPan
  
- `mousemove`:
  - If dragging node → moveDrag(dx, dy), constrain distances
  - If dragging box → moveBox
  - If panning → pan(dx, dy)
  - If hover → update node info
  
- `mouseup`:
  - End drag/pan, saveHistory()

### Keyboard Shortcuts
- `Ctrl+Z` / `Cmd+Z` → undo()
- `Ctrl+Y` / `Cmd+Y` → redo()
- `Space` → toggle play
- `ArrowLeft/Right` → prev/next frame
- `Delete/Backspace` → delete current frame (if > 1)
- `Ctrl+S` → open save modal
- `Ctrl+Shift+S` → export JSON

## Data Flow

```
User Input → interaction.js → mutate state → history.saveHistory()
                                 ↓
                         renderer.js (reads state)
                                 ↓
                         Canvas updated
```

No module besides `interaction.js` should directly mutate nodes/state.
All mutations go through interaction functions.

## Risks & Mitigations

| Risk | Impact | Mitigation |
|------|--------|------------|
| Circular dependencies | High | Follow dependency order: state → (renderer, interaction, animation, hierarchy) → (ui) → app |
| Missing feature in renderer | Medium | Reference original app.js render function line-by-line |
| Event wiring lost | Medium | Copy original event listeners carefully, map to new modules |
| History desync | High | Ensure all mutations call saveHistory() exactly once |
| Performance (thumbnails) | Low | Use offscreen canvas, cache thumbnails |
| Tutorial system breakage | Medium | Import and call as before, no changes to tutorial  |
